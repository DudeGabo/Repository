local FileName = "GaboHubConfig.json"
local SavedSettings = {}

local function Save()
    writefile(FileName, HttpService:JSONEncode(SavedSettings))
end

local function Load()
    if isfile(FileName) then
        local success, data = pcall(function()
            return HttpService:JSONDecode(readfile(FileName))
        end)
        if success and type(data) == "table" then
            for k, v in pairs(data) do
                SavedSettings[k] = v
            end
        end
    end
    
    if SavedSettings.MenuKey == nil then
        SavedSettings.MenuKey = "K"
    end
end

local Antisentrylocals = {
    Players = game:GetService("Players"),
    RunService = game:GetService("RunService"),
    player = game:GetService("Players").LocalPlayer,
    active = false,
    dist = 100,
    pull = -5,
    delay = 3.7,
    target = nil,
    connection = nil,
    detectionTime = 0,
    lastSearch = 0
}

local function getCharacter()
    return Antisentrylocals.player.Character
end

local function equipAndAttack()
    local char = getCharacter()
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local bat = Antisentrylocals.player.Backpack:FindFirstChild("Bat") or char:FindFirstChild("Bat")
    
    if hum and bat then
        if bat.Parent ~= char then
            hum:EquipTool(bat)
        end
        bat:Activate()
      task.wait(0.7)
        hum:UnequipTools()
        
    end
end

local function findTarget()
    local char = getCharacter()
    if not char or not char:FindFirstChild("HumanoidRootPart") then return end
    local rootPos = char.HumanoidRootPart.Position
    for _, obj in ipairs(workspace:GetChildren()) do
        if obj.Name:find("Sentry") and not obj.Name:lower():find("bullet") then
            local ownerId = obj.Name:match("Sentry_(%d+)")
            if ownerId and tonumber(ownerId) == Antisentrylocals.player.UserId then continue end
            local part = obj:IsA("BasePart") and obj or obj:IsA("Model") and (obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart"))
            if part and (rootPos - part.Position).Magnitude <= Antisentrylocals.dist then
                return obj
            end
        end
    end
end

local function moveTarget(obj)
    local char = getCharacter()
    if not char or not char:FindFirstChild("HumanoidRootPart") then return end
    for _, part in ipairs(obj:GetDescendants()) do
        if part:IsA("BasePart") then part.CanCollide = false end
    end
    local root = char.HumanoidRootPart
    local cf = root.CFrame * CFrame.new(0, 0, Antisentrylocals.pull)
    if obj:IsA("BasePart") then
        obj.CFrame = cf
    elseif obj:IsA("Model") then
        local main = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
        if main then main.CFrame = cf end
    end
end

local function start()
    if Antisentrylocals.connection then return end
    Antisentrylocals.connection = Antisentrylocals.RunService.Heartbeat:Connect(function()
        if not Antisentrylocals.active then return end
        if Antisentrylocals.target and Antisentrylocals.target.Parent == workspace then
            if os.clock() - Antisentrylocals.detectionTime >= Antisentrylocals.delay then
                moveTarget(Antisentrylocals.target)
                equipAndAttack()
            end
        else
            if os.clock() - Antisentrylocals.lastSearch >= 0.5 then
                Antisentrylocals.lastSearch = os.clock()
                local found = findTarget()
                if found then
                    Antisentrylocals.target = found
                    Antisentrylocals.detectionTime = os.clock()
                end
            end
        end
    end)
end

local function stop()
    if Antisentrylocals.connection then 
        Antisentrylocals.connection:Disconnect() 
        Antisentrylocals.connection = nil 
    end
    Antisentrylocals.target = nil
end

Antisentrylocals.player.CharacterAdded:Connect(function()
    task.wait(0.5)
    if Antisentrylocals.active then start() else stop() end
end)

if getconnections and aimbot.getconstants then
    for _, r in ipairs(aimbot.netFolder:GetChildren()) do
        if r:IsA("RemoteEvent") then
            local ok, conns = pcall(getconnections, r.OnClientEvent)
            if ok then
                for _, c in ipairs(conns) do
                    if type(c.Function) == "function" then
                        local okc, consts = pcall(aimbot.getconstants, c.Function)
                        if okc then
                            for _, k in ipairs(consts) do
                                if k == "PaintballHitted" then
                                    aimbot.useItemRemote = r
                                    break
                                end
                            end
                        end
                    end
                    if aimbot.useItemRemote then break end
                end
            end
        end
        if aimbot.useItemRemote then break end
    end
end

aimbot.targetRemote = aimbot.useItemRemote and cloneref(aimbot.useItemRemote) or nil
aimbot.fire = aimbot.targetRemote and clonefunction(aimbot.targetRemote.FireServer) or function() end

aimbot.getNearestPlayer = function(maxRange)
    maxRange = maxRange or math.huge
    if not aimbot.player.Character then return nil end
    if not aimbot.HRP then return nil end
    local myPos = aimbot.HRP.Position
    local nearest = nil
    local shortest = maxRange
    for _, pl in ipairs(aimbot.Players:GetPlayers()) do
        if pl ~= aimbot.player and pl.Character and pl.Character:FindFirstChild("HumanoidRootPart") then
            local hrp = pl.Character.HumanoidRootPart
            local dist = (hrp.Position - myPos).Magnitude
            if dist < shortest then
                shortest = dist
                nearest = pl
            end
        end
    end
    return nearest
end

aimbot.hasTool = function(toolName)
    return (aimbot.backpack and aimbot.backpack:FindFirstChild(toolName) ~= nil) or (aimbot.char and aimbot.char:FindFirstChild(toolName) ~= nil)
end

aimbot.equipTool = function(toolName)
    local tool = (aimbot.backpack and aimbot.backpack:FindFirstChild(toolName)) or (aimbot.char and aimbot.char:FindFirstChild(toolName))
    if tool and aimbot.backpack and tool.Parent == aimbot.backpack then
        tool.Parent = aimbot.char
        return true
    end
    return tool ~= nil
end

aimbot.equipOnlyTool = function(toolName)
    if aimbot.char and aimbot.backpack then
        for _, tool in ipairs(aimbot.char:GetChildren()) do
            if tool:IsA("Tool") and tool.Name ~= toolName then
                tool.Parent = aimbot.backpack
            end
        end
    end
    return aimbot.equipTool(toolName)
end

aimbot.useLaserCape = function(targetPart)
    if not targetPart or not aimbot.targetRemote then return end
    local args = {targetPart.Position, targetPart}
    pcall(function()
        aimbot.fire(aimbot.targetRemote, unpack(args))
    end)
end

aimbot.useWebSlinger = function(targetPart)
    if not targetPart or not aimbot.targetRemote then return end
    local tool = (aimbot.backpack and aimbot.backpack:FindFirstChild("Web Slinger")) or (aimbot.char and aimbot.char:FindFirstChild("Web Slinger"))
    if tool and tool:FindFirstChild("Handle") then
        local handle = tool.Handle
        local args = {
            Vector3.new(targetPart.Position.X, targetPart.Position.Y, targetPart.Position.Z),
            targetPart,
            handle
        }
        pcall(function()
            aimbot.fire(aimbot.targetRemote, unpack(args))
        end)
    end
end

aimbot.setupLaserAim = function()
    local laserTool = (aimbot.backpack and aimbot.backpack:FindFirstChild("Laser Cape")) or (aimbot.char and aimbot.char:FindFirstChild("Laser Cape"))
    if not laserTool then return end
    
    if aimbot.laserConnection then
        aimbot.laserConnection:Disconnect()
        aimbot.laserConnection = nil
    end
    
    aimbot.laserConnection = laserTool.Activated:Connect(function()
        local target = aimbot.getNearestPlayer(aimbot.range)
        if target and target.Character then
            local targetPart = target.Character:FindFirstChild("HumanoidRootPart")
            if targetPart then
                aimbot.useLaserCape(targetPart)
            end
        end
    end)
end

aimbot.setupWebAim = function()
    local webTool = (aimbot.backpack and aimbot.backpack:FindFirstChild("Web Slinger")) or (aimbot.char and aimbot.char:FindFirstChild("Web Slinger"))
    
    if not webTool then
        local args = {"Web Slinger"}
        pcall(function()
            aimbot.ReplicatedStorage:WaitForChild("Packages"):WaitForChild("Net"):WaitForChild("RF/CoinsShopService/RequestBuy"):InvokeServer(unpack(args))
        end)
        task.wait(1)
        webTool = (aimbot.backpack and aimbot.backpack:FindFirstChild("Web Slinger")) or (aimbot.char and aimbot.char:FindFirstChild("Web Slinger"))
    end
    
    if not webTool then return end
    
    if aimbot.webConnection then
        aimbot.webConnection:Disconnect()
        aimbot.webConnection = nil
    end
    
    aimbot.webConnection = webTool.Activated:Connect(function()
        local target = aimbot.getNearestPlayer(aimbot.range)
        if target and target.Character then
            local targetPart = target.Character:FindFirstChild("HumanoidRootPart")
            if targetPart then
                aimbot.useWebSlinger(targetPart)
            end
        end
    end)
end

            end
        end
    end
end

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local LocalPlayer = Players.LocalPlayer

local antirag = {
    mode = nil,
    connections = {},
    charData = {}
}

local function cacheCharacterData()
    local char = LocalPlayer.Character
    if not char then return false end
    
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    
    if not hum or not root then return false end
    
    antirag.charData = {
        character = char,
        humanoid = hum,
        root = root,
        originalWalkSpeed = hum.WalkSpeed,
        originalJumpPower = hum.JumpPower,
        isFrozen = false
    }
    
    return true
end

local function disconnectAll()
    for _, conn in ipairs(antirag.connections) do
        if typeof(conn) == "RBXScriptConnection" then
            pcall(function() conn:Disconnect() end)
        end
    end
    antirag.connections = {}
end

local function isRagdolled()
    if not antirag.charData.humanoid then return false end
    
    local hum = antirag.charData.humanoid
    local state = hum:GetState()
    
    local ragdollStates = {
        [Enum.HumanoidStateType.Physics] = true,
        [Enum.HumanoidStateType.Ragdoll] = true,
        [Enum.HumanoidStateType.FallingDown] = true
    }
    
    if ragdollStates[state] then
        return true
    end
    
    local endTime = LocalPlayer:GetAttribute("RagdollEndTime")
    if endTime then
        local now = workspace:GetServerTimeNow()
        if (endTime - now) > 0 then
            return true
        end
    end
    
    return false
end

local function removeRagdollConstraints()
    if not antirag.charData.character then return end
    
    local removed = false
    
    for _, descendant in ipairs(antirag.charData.character:GetDescendants()) do
        if descendant:IsA("BallSocketConstraint") or 
           (descendant:IsA("Attachment") and descendant.Name:find("RagdollAttachment")) then
            pcall(function()
                descendant:Destroy()
                removed = true
            end)
        end
    end
    
    return removed
end

local function forceExitRagdoll()
    if not antirag.charData.humanoid or not antirag.charData.root then return end
    
    local hum = antirag.charData.humanoid
    local root = antirag.charData.root
    
    pcall(function()
        local now = workspace:GetServerTimeNow()
        LocalPlayer:SetAttribute("RagdollEndTime", now)
    end)
    
    if hum.Health > 0 then
        hum:ChangeState(Enum.HumanoidStateType.Running)
    end
    
    root.Anchored = false
    root.AssemblyLinearVelocity = Vector3.zero
    root.AssemblyAngularVelocity = Vector3.zero
end

local function freezePlayer()
    if not antirag.charData.humanoid or not antirag.charData.root then return end
    if antirag.charData.isFrozen then return end
    
    local hum = antirag.charData.humanoid
    local root = antirag.charData.root
    
    antirag.charData.originalWalkSpeed = hum.WalkSpeed
    antirag.charData.originalJumpPower = hum.JumpPower
    
    hum.WalkSpeed = 0
    hum.JumpPower = 0
    root.Anchored = true
    root.AssemblyLinearVelocity = Vector3.zero
    root.AssemblyAngularVelocity = Vector3.zero
    
    antirag.charData.isFrozen = true
end

local function unfreezePlayer()
    if not antirag.charData.humanoid or not antirag.charData.root then return end
    if not antirag.charData.isFrozen then return end
    
    local hum = antirag.charData.humanoid
    local root = antirag.charData.root
    
    hum.WalkSpeed = antirag.charData.originalWalkSpeed or 16
    hum.JumpPower = antirag.charData.originalJumpPower or 50
    
    root.Anchored = false
    root.AssemblyLinearVelocity = Vector3.zero
    root.AssemblyAngularVelocity = Vector3.zero
    
    antirag.charData.isFrozen = false
end

local function v1HeartbeatLoop()
    while antirag.mode == "v1" and antirag.charData.humanoid do
        task.wait()
        
        if isRagdolled() then
            removeRagdollConstraints()
            forceExitRagdoll()
        end
    end
end

local function v2HeartbeatLoop()
    while antirag.mode == "v2" and antirag.charData.humanoid do
        task.wait()
        
        if isRagdolled() then
            freezePlayer()
        else
            unfreezePlayer()
        end
    end
end

local function setupCameraBinding()
    if not antirag.charData.humanoid then return end
    
    local conn = RunService.RenderStepped:Connect(function()
        if antirag.mode ~= "v1" then return end
        
        local cam = workspace.CurrentCamera
        if cam and antirag.charData.humanoid and cam.CameraSubject ~= antirag.charData.humanoid then
            cam.CameraSubject = antirag.charData.humanoid
        end
    end)
    
    table.insert(antirag.connections, conn)
end

local function onCharacterAdded(char)
    task.wait(0.5)
    
    if not antirag.mode then return end
    
    if cacheCharacterData() then
        if antirag.mode == "v1" then
            setupCameraBinding()
            task.spawn(v1HeartbeatLoop)
        elseif antirag.mode == "v2" then
            task.spawn(v2HeartbeatLoop)
        end
    end
end

local function EnableAntiRagdoll(mode)
    if mode ~= "v1" and mode ~= "v2" then return end
    if antirag.mode == mode then return end
    
    if antirag.mode then
        if antirag.charData.isFrozen then
            unfreezePlayer()
        end
        disconnectAll()
        antirag.charData = {}
    end
    
    if not cacheCharacterData() then
        return
    end
    
    antirag.mode = mode
    
    local charConn = LocalPlayer.CharacterAdded:Connect(onCharacterAdded)
    table.insert(antirag.connections, charConn)
    
    if mode == "v1" then
        setupCameraBinding()
        task.spawn(v1HeartbeatLoop)
    else
        task.spawn(v2HeartbeatLoop)
    end
end

local function DisableAntiRagdoll()
    if not antirag.mode then return end
    antirag.mode = nil
    
    if antirag.charData.isFrozen then
        unfreezePlayer()
    end
    
    disconnectAll()
    antirag.charData = {}
end

createToggle(moveSec, "Anti Ragdoll", function(active)
    if active then
        EnableAntiRagdoll("v1")
    else
        DisableAntiRagdoll()
    end
end)

    for _, desc in pairs(adminPanel:GetDescendants()) do
        if desc:IsA("TextButton") or desc:IsA("ImageButton") then

    for _, base in pairs(plots:GetChildren()) do

            for _, d in pairs(base:GetDescendants()) do
                if d:IsA("TextLabel") and (string.find(d.Text, locals.lp.Name) or string.find(d.Text, locals.lp.DisplayName)) then

        if p ~= locals.lp and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local dist = (p.Character.HumanoidRootPart.Position - basePos).Magnitude
            table.insert(targets, {player = p, distance = dist})
        end
    end
    
    table.sort(targets, function(a, b) return a.distance < b.distance end)
    
    if #targets == 0 then return end
    
    local now = os.time()
    local balloonAvailable = (now - locals.lastBalloon >= 30)
    local ragdollAvailable = (now - locals.lastRagdoll >= 30)
    
    local cmdButtons = getCommandButtonsFromPanel()
    if #cmdButtons == 0 then return end
    
    if #targets >= 2 then
        if balloonAvailable then
            local btn = findPlayerButtonInPanel(targets[1].player)
            if btn then
                runSingleCommand(btn, cmdButtons, "balloon")
                locals.lastBalloon = now
            end
        end
        if ragdollAvailable then
            local btn = findPlayerButtonInPanel(targets[2].player)
            if btn then
                runSingleCommand(btn, cmdButtons, "ragdoll")
                locals.lastRagdoll = now
            end
        end
    elseif #targets == 1 then
        if balloonAvailable then
            local btn = findPlayerButtonInPanel(targets[1].player)
            if btn then
                runSingleCommand(btn, cmdButtons, "balloon")
                locals.lastBalloon = now
            end
        elseif ragdollAvailable then
            local btn = findPlayerButtonInPanel(targets[1].player)
            if btn then
                runSingleCommand(btn, cmdButtons, "ragdoll")
                locals.lastRagdoll = now
            end
        end
    end
end

        local purchases = myBase:FindFirstChild("Purchases")

        if timeLabel then
            local digit = tonumber(timeLabel.Text:match("%d+"))
            if digit then
                lastKnownTime = digit
                lastUpdateTick = tick()
            end
        end
    end

    if lastKnownTime then
        local timePassed = tick() - lastUpdateTick
        local estimatedTime = lastKnownTime - timePassed

        if estimatedTime <= 5.1 and estimatedTime > 4.5 and not hasWarned then
            ShowWarning(5)
            hasWarned = true
        end

        if estimatedTime <= 2 then
            ExecuteRejoin()
        end
        
        if estimatedTime > 6 then
            hasWarned = false
            rejoinProcessed = false
        end
    end
end)

            if data.selectionBox and data.selectionBox.Parent then data.selectionBox:Destroy() end
            if data.billboardGui and data.billboardGui.Parent then data.billboardGui:Destroy() end
            minelocal.ESPData[i] = nil
        end
        return
    end

    local toolsFolder = minelocal.Workspace:FindFirstChild(minelocal.FolderName)
    if not toolsFolder then return end

    local currentMines = {}
    for _, obj in pairs(toolsFolder:GetChildren()) do
        if obj.Name:match("^SubspaceTripmine") and obj:IsA("BasePart") then
            currentMines[obj] = true
            if not minelocal.ESPData[obj] then 
                minelocal.ESPData[obj] = minelocal.createMineESP(obj) 
            end
        end
    end

    for mineObj, data in pairs(minelocal.ESPData) do
        if not currentMines[mineObj] or not mineObj.Parent then
            if data.selectionBox and data.selectionBox.Parent then data.selectionBox:Destroy() end
            if data.billboardGui and data.billboardGui.Parent then data.billboardGui:Destroy() end
            minelocal.ESPData[mineObj] = nil
        end
    end

local function updateBaseESP()
    if not plotsFolder then return end

    for _, plot in ipairs(plotsFolder:GetChildren()) do
        local purchases = plot:FindFirstChild("Purchases")
        local plotBlock = purchases and purchases:FindFirstChild("PlotBlock")
        local mainPart = plotBlock and plotBlock:FindFirstChild("Main")
        local billboard = baseEspInstances[plot.Name]

        local timeLabel = mainPart
            and mainPart:FindFirstChild("BillboardGui")
            and mainPart.BillboardGui:FindFirstChild("RemainingTime")

        if timeLabel and mainPart then
            billboard = billboard or createBaseESP(plot, mainPart)
            local label = billboard:FindFirstChildWhichIsA("TextLabel")
            
            if label then
                local t = timeLabel.Text
                local isNumber = string.match(t, "%d")
                
                if not isNumber or t == "0" or t == "00:00" or t == "0s" or t == "" or t == "Unlock" or t == "Locked Base" or t == "Lock Base" then
                    label.Text = "Unlock"
                elseif t == "1" then 
                    label.Text = "01"
                else
                    label.Text = t
                end
            end
        elseif billboard then
            if billboard.Adornee and billboard.Adornee.Name == "EspOffset" then
                billboard.Adornee:Destroy()
            end
            billboard:Destroy()
            baseEspInstances[plot.Name] = nil
        end
    end
end

function EspBest.ParseMoneyValue(text)
    text = string.lower(text or "")
    local num = tonumber(text:match("[%d%.]+")) or 0
    if text:find("k") then num = num * 1e3
    elseif text:find("m") then num = num * 1e6
    elseif text:find("b") then num = num * 1e9
    elseif text:find("t") then num = num * 1e12
    elseif text:find("qa") then num = num * 1e15
    end
    return num
end

function EspBest.ReadDisplayAndGenerationData(part)
    local displayName, generation
    for _, child in ipairs(part:GetChildren()) do
        if child:IsA("BillboardGui") or child:IsA("SurfaceGui") then
            for _, label in ipairs(child:GetDescendants()) do
                if label:IsA("TextLabel") then
                    if label.Name == "DisplayName" then
                        displayName = label.Text
                    elseif label.Name == "Generation" then
                        generation = label.Text
                    end
                end
            end
        end
    end
    return displayName, generation
end

function EspBest.FindAllAnimalsInWorkspace()
    local results = {}
    local step = 0
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local displayName, generation = EspBest.ReadDisplayAndGenerationData(obj)
            if displayName and displayName ~= "" and generation and generation ~= "" then
                table.insert(results, {
                    part = obj,
                    displayName = displayName,
                    generation = generation,
                    value = EspBest.ParseMoneyValue(generation),
                    position = obj.Position,
                })
            end
        end
        step = step + 1
        if step % 800 == 0 then task.wait() end
    end
    return results
end

function EspBest.ExtractPodiumNumber(podium)
    local n = tonumber(podium.Name)
    if n then return n end
    local m = podium.Name:match("(%d+)")
    if m then return tonumber(m) end
    return nil
end

function EspBest.GetPodiumWorldPosition(podium)
    if podium:IsA("BasePart") then return podium.Position end
    if podium:IsA("Model") and podium.PrimaryPart then return podium.PrimaryPart.Position end
    local part = podium:FindFirstChildWhichIsA("BasePart", true)
    if part then return part.Position end
    if podium:IsA("Model") then
        local ok, cf = pcall(function() return podium:GetPivot().Position end)
        if ok then return cf end
    end
    return nil
end

function EspBest.FindPodiumsInsidePlot(plot)
    local results, seen = {}, {}
    for _, desc in ipairs(plot:GetDescendants()) do
        if (desc:IsA("Folder") or desc:IsA("Model")) and desc.Name:lower():find("podium")
            and not tonumber(desc.Name) then
            for _, child in ipairs(desc:GetChildren()) do
                local num = EspBest.ExtractPodiumNumber(child)
                if num and not seen[child] then
                    seen[child] = true
                    local pos = EspBest.GetPodiumWorldPosition(child)
                    if pos then
                        table.insert(results, { podium = child, number = num, position = pos })
                    end
                end
            end
        end
    end
    if #results == 0 then
        local numbered = {}
        for _, desc in ipairs(plot:GetDescendants()) do
            if (desc:IsA("Model") or desc:IsA("Folder")) and tonumber(desc.Name) then
                local num = tonumber(desc.Name)
                if num and num >= 1 and num <= 20 then
                    table.insert(numbered, { obj = desc, num = num })
                end
            end
        end
        if #numbered >= 3 then
            for _, item in ipairs(numbered) do
                if not seen[item.obj] then
                    seen[item.obj] = true
                    local pos = EspBest.GetPodiumWorldPosition(item.obj)
                    if pos then
                        table.insert(results, { podium = item.obj, number = item.num, position = pos })
                    end
                end
            end
        end
    end
    return results
end

function EspBest.DeterminePlotOwnerName(plot)
    local plotSign = plot:FindFirstChild("PlotSign", true)
    if plotSign then
        for _, desc in ipairs(plotSign:GetDescendants()) do
            if desc:IsA("TextLabel") and desc.Text and desc.Text ~= "" then
                local text = desc.Text
                local uppercaseText = text:upper()
                if uppercaseText == "YOUR BASE" then return "YOU" end
                if text:lower():find("empty") then return "Empty Base" end
                local m = text:match("[Bb]ase [Oo]f%s+(.+)")
                if m then return m end
                if #text > 0 and #text < 30 then return text end
            end
        end
    end
    return plot.Name
end

function EspBest.CalculatePlotBoundaries(plot)
    local minX, maxX = math.huge, -math.huge
    local minZ, maxZ = math.huge, -math.huge
    local found = false
    for _, desc in ipairs(plot:GetDescendants()) do
        if desc:IsA("BasePart") then
            local pos = desc.Position
            if pos.X < minX then minX = pos.X end
            if pos.X > maxX then maxX = pos.X end
            if pos.Z < minZ then minZ = pos.Z end
            if pos.Z > maxZ then maxZ = pos.Z end
            found = true
        end
    end
    if not found then return nil end
    return { minX = minX, maxX = maxX, minZ = minZ, maxZ = maxZ }
end

function EspBest.FilterAnimalsByPlotArea(animals, plot, bounds)
    local nearby, padding = {}, 6 * ScaleFactor
    for _, animal in ipairs(animals) do
        local inside = animal.part:IsDescendantOf(plot)
        if not inside and bounds then
            local p = animal.position
            inside = p.X >= bounds.minX - padding and p.X <= bounds.maxX + padding
                and p.Z >= bounds.minZ - padding and p.Z <= bounds.maxZ + padding
        end
        if inside then table.insert(nearby, animal) end
    end
    return nearby
end

function EspBest.MatchAnimalsToClosestPodiums(podiums, animals)
    local candidates = {}
    for podiumIndex, p in ipairs(podiums) do
        for animalIndex, animal in ipairs(animals) do
            local diff = animal.position - p.position
            local horizDist = math.sqrt(diff.X * diff.X + diff.Z * diff.Z)
            local yDist = math.abs(diff.Y)
            if horizDist < PODIUM_MATCH_DISTANCE and yDist < 60 * ScaleFactor then
                local score = math.sqrt(horizDist * horizDist + yDist * yDist * 0.15)
                table.insert(candidates, { pi = podiumIndex, ai = animalIndex, score = score })
            end
        end
    end
    table.sort(candidates, function(a, b) return a.score < b.score end)

    local podiumToAnimal, usedAnimals, usedPodiums = {}, {}, {}
    for _, c in ipairs(candidates) do
        if not usedPodiums[c.pi] and not usedAnimals[c.ai] then
            podiumToAnimal[c.pi] = c.ai
            usedPodiums[c.pi] = true
            usedAnimals[c.ai] = true
        end
    end

    local entries = {}
    for podiumIndex, p in ipairs(podiums) do
        local matchedIndex = podiumToAnimal[podiumIndex]
        if matchedIndex then
            local a = animals[matchedIndex]
            table.insert(entries, {
                number = p.number,
                podium = p.podium,
                podiumPosition = p.position,
                name = a.displayName,
                generation = a.generation,
                value = a.value,
                animalPart = a.part,
                animalPosition = a.position,
            })
        end
    end
    return entries
end

function EspBest.ScanForAnimals()
    local animals = EspBest.FindAllAnimalsInWorkspace()
    local plots = EspBest.FindAllPlotsFolder()
    local entries = {}
    for _, plot in ipairs(plots) do
        local isMine = false
        pcall(function() isMine = false end)
        if not isMine then
            local owner = EspBest.DeterminePlotOwnerName(plot)
            local podiums = EspBest.FindPodiumsInsidePlot(plot)
            local bounds = EspBest.CalculatePlotBoundaries(plot)
            local plotAnimals = EspBest.FilterAnimalsByPlotArea(animals, plot, bounds)
            local matched = EspBest.MatchAnimalsToClosestPodiums(podiums, plotAnimals)
            for _, e in ipairs(matched) do
                e.plot = plot
                e.plotName = owner
                e.uid = tostring(e.number) .. "_" .. tostring(e.name)
                table.insert(entries, e)
            end
        end
    end
    table.sort(entries, function(a, b)
        return (a.value or 0) > (b.value or 0)
    end)
    return entries
end

local FOV_MANAGER = {
    activeCount = 0,
    conn = nil,
    forcedFOV = 70,
}

function FOV_MANAGER:Start()
    if self.conn then return end
    self.conn = RunService.RenderStepped:Connect(function()
        local cam = Workspace.CurrentCamera
        if cam and cam.FieldOfView ~= self.forcedFOV then
            cam.FieldOfView = self.forcedFOV
        end
    end)
end

function FOV_MANAGER:Stop()
    if self.conn then
        self.conn:Disconnect()
        self.conn = nil
    end
end

function FOV_MANAGER:Push()
    self.activeCount = self.activeCount + 1
    self:Start()
end

function FOV_MANAGER:Pop()
    if self.activeCount > 0 then
        self.activeCount = self.activeCount - 1
    end
    if self.activeCount == 0 then
        self:Stop()
    end
end

local ANTI_BEE_DISCO = {}
local antiBeeDiscoRunning = false
local antiBeeDiscoConnections = {}
local originalMoveFunction = nil
local controlsProtected = false

local BAD_LIGHTING_NAMES = {
    Blue = true,
    DiscoEffect = true,
    BeeBlur = true,
    ColorCorrection = true,
}

local function antiBeeDiscoNuke(obj)
    if not obj or not obj.Parent then return end
    if BAD_LIGHTING_NAMES[obj.Name] then
        pcall(function()
            obj:Destroy()
        end)
    end
end

local function antiBeeDiscoDisconnectAll()
    for _, conn in ipairs(antiBeeDiscoConnections) do
        if typeof(conn) == "RBXScriptConnection" then
            conn:Disconnect()
        end
    end
    antiBeeDiscoConnections = {}
end

local function protectControls()
    if controlsProtected then return end
    pcall(function()
        local PlayerScripts = lp.PlayerScripts
        local PlayerModule = PlayerScripts:FindFirstChild("PlayerModule")
        if not PlayerModule then return end
        local Controls = require(PlayerModule):GetControls()
        if not Controls then return end
        if not originalMoveFunction then
            originalMoveFunction = Controls.moveFunction
        end
        local function protectedMoveFunction(self, moveVector, relativeToCamera)
            if originalMoveFunction then
                originalMoveFunction(self, moveVector, relativeToCamera)
            end
        end
        local controlCheckConn = RunService.Heartbeat:Connect(function()
            if not antiBeeDiscoRunning then return end
            if Controls.moveFunction ~= protectedMoveFunction then
                Controls.moveFunction = protectedMoveFunction
            end
        end)
        table.insert(antiBeeDiscoConnections, controlCheckConn)
        Controls.moveFunction = protectedMoveFunction
        controlsProtected = true
    end)
end

local function restoreControls()
    if not controlsProtected then return end
    pcall(function()
        local PlayerScripts = lp.PlayerScripts
        local PlayerModule = PlayerScripts:FindFirstChild("PlayerModule")
        if not PlayerModule then return end
        local Controls = require(PlayerModule):GetControls()
        if not Controls or not originalMoveFunction then return end
        Controls.moveFunction = originalMoveFunction
        controlsProtected = false
    end)
end

local function blockBuzzingSound()
    pcall(function()
        local PlayerScripts = lp.PlayerScripts
        local beeScript = PlayerScripts:FindFirstChild("Bee", true)
        if beeScript then
            local buzzing = beeScript:FindFirstChild("Buzzing")
            if buzzing and buzzing:IsA("Sound") then
                buzzing:Stop()
                buzzing.Volume = 0
            end
        end
    end)
end

function ANTI_BEE_DISCO.Enable()
    if antiBeeDiscoRunning then return end
    antiBeeDiscoRunning = true
    for _, inst in ipairs(Lighting:GetDescendants()) do
        antiBeeDiscoNuke(inst)
    end
    table.insert(antiBeeDiscoConnections, Lighting.DescendantAdded:Connect(function(obj)
        if not antiBeeDiscoRunning then return end
        antiBeeDiscoNuke(obj)
    end))
    protectControls()
    table.insert(antiBeeDiscoConnections, RunService.Heartbeat:Connect(function()
        if not antiBeeDiscoRunning then return end
        blockBuzzingSound()
    end))
    FOV_MANAGER:Push()
end

function ANTI_BEE_DISCO.Disable()
    if not antiBeeDiscoRunning then return end
    antiBeeDiscoRunning = false
    restoreControls()
    antiBeeDiscoDisconnectAll()
    FOV_MANAGER:Pop()
end

createToggle(otherSecVision, "Anti Effects", function(state)
    if state then
        ANTI_BEE_DISCO.Enable()
    else
        ANTI_BEE_DISCO.Disable()
    end
end)

local function stopAllAnimations()
    for _, item in ipairs(workspace:GetDescendants()) do
        if item:IsA("Humanoid") or item:IsA("AnimationController") then
            if not (lp.Character and item:IsDescendantOf(lp.Character)) then
                local animator = item:FindFirstChildOfClass("Animator")
                if animator then
                    for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
                        track:Stop(0)
                    end
                end
            end
        end
    end
end
